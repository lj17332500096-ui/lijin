```python
import time

print("开始 5 秒死循环...")
start = time.time()
iteration = 0

while True:
    iteration += 1
    elapsed = time.time() - start
    
    if elapsed >= 5:
        break
    
    # 模拟一些工作
    _ = sum(range(1000))
    
    if iteration % 10000 == 0:
        print(f"  迭代 {iteration}, 已运行 {elapsed:.2f} 秒")

print(f"\n完成！总共迭代 {iteration} 次，用时 {time.time() - start:.2f} 秒")
```
