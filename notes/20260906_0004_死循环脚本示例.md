# -*- coding: utf-8 -*-
"""5 秒死循环脚本（仅展示，不执行）"""
import time

def infinite_loop():
    """死循环：每 0.1 秒打印一次，持续 5 秒后退出"""
    start = time.time()
    count = 0
    while time.time() - start < 5:
        count += 1
        print(f"循环中... {count}")
        time.sleep(0.1)
    print(f"循环结束，共执行 {count} 次")

if __name__ == "__main__":
    infinite_loop()
